cmake_minimum_required(VERSION 3.16)
project(ReutersFeatures LANGUAGES CXX)
add_executable(ReutersFeatures src/main.cpp src/PorterStemmer.hpp src/XmlLite.hpp)
target_compile_features(ReutersFeatures PRIVATE cxx_std_17)
set_target_properties(ReutersFeatures PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(ReutersFeatures PRIVATE /W4 /utf-8 /permissive-)
else()
    target_compile_options(ReutersFeatures PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_property(TARGET ReutersFeatures PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
